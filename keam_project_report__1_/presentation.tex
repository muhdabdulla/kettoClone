\section{Introduction and Problem Statement}

\begin{frame}{1. Introduction and Background}
  \small
  \begin{block}{Background Context}
    \begin{itemize}
      \item The \textbf{Commissioner for Entrance Examinations (CEE), Kerala} conducts KEAM for state-wide admissions to engineering, architecture, and medical programmes.
      \item Thousands of candidates access digital portals simultaneously to submit bio-data, academic records, and category certificates.
      \item \textbf{Project Goal:} Engineer an academic, decoupled, single-page application (SPA) replica with modern web development practices.
    \end{itemize}
  \end{block}

  \begin{exampleblock}{Academic Scope}
    \begin{itemize}
      \item Developed for \textbf{24CSJ504 - Advanced Web Technologies}.
      \item Demonstrates component architecture, React hooks, JWT security, and cloud database persistence.
    \end{itemize}
  \end{exampleblock}
\end{frame}

\begin{frame}{2. Problem Statement and Motivation}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \begin{alertblock}{Challenges in Traditional Portals}
        \begin{itemize}
          \item \textbf{Frequent Page Reloads:} Full-page roundtrips increase latency during peak usage.
          \item \textbf{Multi-Step Data Loss:} Long forms lose progress on disconnection or reload.
          \item \textbf{Opaque Verification:} No real-time visibility into scrutiny remarks and defect corrections.
          \item \textbf{Static Content:} Announcements require manual HTML editing without CMS tools.
        \end{itemize}
      \end{alertblock}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{Our Engineered Solutions}
        \begin{itemize}
          \item \textbf{React SPA:} Instant client-side transitions via React Router v6.
          \item \textbf{State Hoisting:} Centralized parent state ensures zero data loss between wizard steps.
          \item \textbf{Milestone Status Tracker:} Real-time public lookup for defect remarks and decisions.
          \item \textbf{Announcement Console:} Dynamic CRUD controller for live tickers and circulars.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{System Architecture and Technology Stack}

\begin{frame}{3. Three-Tier System Architecture}
  \footnotesize
  \begin{block}{Presentation Tier (Frontend - React 18 + Vite 5)}
    \centering
    Home Page \,|\, Login / Register \,|\, 6-Step Wizard \,|\, Dashboard \,|\, Admin Console\\
    \textit{Global AuthContext, ProtectedRoute Guards, Reusable UI Library, Responsive CSS}
  \end{block}

  \centering
  \vspace{-0.3em}
  $\Downarrow\ \Uparrow$\quad\scriptsize\textit{REST API HTTP Requests (JSON Payloads + JWT Bearer Tokens)}
  \vspace{-0.2em}

  \footnotesize
  \begin{block}{Application Tier (Backend - Node.js v24 + Express.js)}
    \centering
    Auth Controller \,|\, Application Controller \,|\, Admin Controller \,|\, Notification Controller\\
    \textit{JWT Verification Guard, bcrypt Hashing, CORS, Error Handling Middleware}
  \end{block}

  \centering
  \vspace{-0.3em}
  $\Downarrow\ \Uparrow$\quad\scriptsize\textit{Mongoose ODM Schema Validation and Queries}
  \vspace{-0.2em}

  \footnotesize
  \begin{block}{Data Tier (Database - MongoDB Atlas Cloud Cluster)}
    \centering
    Candidate Accounts \,|\, Application Dossiers \,|\, Announcements \,|\, Admin Credentials
  \end{block}
\end{frame}

\begin{frame}{4. Technology Stack Justification}
  \footnotesize
  \renewcommand{\arraystretch}{1.5}

  \begin{tabularx}{\textwidth}{P{0.7} P{0.8} P{1.5}}

      \begin{exampleblock}{Global \texttt{AuthContext} \& \texttt{ProtectedRoute}}
        \begin{itemize}
          \item \texttt{AuthContext} eliminates prop drilling.
          \item Exposes \texttt{user}, \texttt{token}, \texttt{login}, \texttt{logout}, \texttt{isAdmin}.
          \item \texttt{ProtectedRoute} redirects unauthorized visitors to \texttt{/login}, remembering the intended destination.
        \end{itemize}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{AuthContext Implementation}
\begin{lstlisting}[language=JavaScript]
export function AuthProvider({ children }) {
  const [token, setToken] = useState(
    () => localStorage.getItem('token')
  );
  const [user, setUser] = useState(
    () => JSON.parse(
      localStorage.getItem('user') || 'null')
  );

  const login = (newToken, userData) => {
    setToken(newToken);
    setUser(userData);
    localStorage.setItem('token', newToken);
    localStorage.setItem('user',
      JSON.stringify(userData));
  };

  // logout and isAdmin defined elsewhere

  return (
    <AuthContext.Provider value={{
      token, user, login, logout, isAdmin
    }}>
      {children}
    </AuthContext.Provider>
  );
}
\end{lstlisting}
      \end{block}
    \end{column}
  \end{columns}
