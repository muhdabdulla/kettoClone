\section{Motivation \& Problem Statement}

\begin{frame}{Motivation \& Problem Statement}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Healthcare Crisis in India}
        \begin{itemize}
          \item Over \textbf{65\% of medical expenditures} in India are financed out-of-pocket.
          \item Pediatric heart surgeries, organ transplants, and cancer therapies require \textbf{₹15L--₹30L within 48 hours}.
          \item Traditional loans entail institutional delays that prove fatal for emergency patients.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{Deficiencies in Conventional Portals}
        \begin{itemize}
          \item \textbf{Heavy Deductions:} 5\% to 12\% platform commission deducted from patient relief.
          \item \textbf{Laggy Page Reloads:} Full-page browser requests disrupt payment checkouts.
          \item \textbf{Opaque Tax Incentives:} Absence of instant Section 80G tax receipt automation.
          \item \textbf{Rigid Giving:} Lack of recurring monthly micro-giving plans.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Code Implementation Highlights}

\begin{frame}[fragile]{Backend: Atomic Donation Transaction Controller}
\begin{lstlisting}[language=JavaScript]
// POST /api/donations - Atomically increments raised amount & donor count
exports.createDonation = async (req, res, next) => {
  try {
    const { campaignId, amount, donorName, donorEmail, panCard } = req.body;
    
    // 1. Record individual donation entry
    const donation = await Donation.create({
      campaignId, amount, donorName, donorEmail, panCard
    });

    // 2. Atomic increment to eliminate race conditions under concurrent load
    const updatedCampaign = await Campaign.findByIdAndUpdate(
      campaignId,
      { $inc: { raisedAmount: Number(amount), donorsCount: 1 } },
      { new: true, runValidators: true }
    );

    return res.status(201).json({
      success: true, message: 'Donation recorded successfully', data: donation
    });
  } catch (err) { next(err); }
};
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Frontend: React State Management with Optimistic Updates}
\begin{lstlisting}[language=JavaScript]
// React Context: Optimistic UI state updates with celebratory confetti
const handleMakeDonation = async ({ campaignId, amount, donorName, donorEmail, panCard }) => {
  // 1. Post to REST API
  await createDonation({ campaignId, amount, donorName, donorEmail, panCard });

  // 2. Optimistic UI update across all active campaign cards
  setCampaigns(prev => prev.map(c => 
    c._id === campaignId 
      ? { ...c, raisedAmount: (c.raisedAmount || 0) + Number(amount), donorsCount: (c.donorsCount || 0) + 1 }
      : c
  ));

  // 3. Generate instant tax receipt entry in user donation history
  const receipt = { id: 'REC_' + Date.now(), campaignId, amount, donorName, date: new Date().toLocaleDateString() };
  setUserDonations(prev => [receipt, ...prev]);

  // 4. Trigger celebration confetti particles
  confetti({ particleCount: 80, spread: 70, origin: { y: 0.6 } });
  return receipt;
};
\end{lstlisting}
\end{frame}
